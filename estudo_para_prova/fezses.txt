int vetor[10]



int * inverter(int vetorRecebido){
    int tamanho = fim = 10;
    int inicio = 0;
    int aux;
    for (int i = 0; i < tamanho; i++){
        int aux = vetorRecebido[inicio];
        vetorRecebido[inicio] = vetorRecebido[fim]
        vetorRecebido[fim] = aux;
        inicio++;
        fim--;
    }
}
